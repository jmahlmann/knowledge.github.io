    Spack Environments¶

    EPaCO systems provide pre-compiled environments for common workflows. Use those when possible:

    source /dartfs/rc/lab/E/EPaCO/activate-entity.sh

    source /dartfs/rc/lab/E/EPaCO/activate-cactus.sh

    If you need to build your own environment, the configurations below list the required components.

    spack:
  specs:
  - gcc@12.5.0 %gcc@8.5.0
  - ucx+cuda %gcc@12.5.0
  - ucx~cuda %gcc@12.5.0
  - openmpi+cuda %gcc@12.5.0
  - openmpi~cuda %gcc@12.5.0
  - lmod %gcc@12.5.0

  view: false

  concretizer:
    unify: false

  packages:
    cuda:
      externals:
      - spec: cuda@13.0.48
        prefix: /optnfs/common/cuda/13.0

    gcc:
      externals:
      - spec: gcc@8.5.0 languages:='c,c++,fortran'
        prefix: /usr
        extra_attributes:
          compilers:
            c: /usr/bin/gcc
            cxx: /usr/bin/g++
            fortran: /usr/bin/gfortran

    spack:
  specs:
  - gcc@12.5.0
  - mpich %gcc@12.5.0
  - hdf5+cxx+fortran+hl+mpi %gcc@12.5.0
  - netlib-lapack %gcc@12.5.0
  - libjpeg %gcc@12.5.0
  - openssl %gcc@12.5.0
  - fftw %gcc@12.5.0
  - zlib %gcc@12.5.0
  - papi %gcc@12.5.0
  - gsl %gcc@12.5.0
  - hwloc %gcc@12.5.0
  - xz %gcc@12.5.0
  - amrex %gcc@12.5.0
  - adios2+blosc+fortran+hdf5+mpi+sst %gcc@12.5.0

  view: true

  concretizer:
    unify: when_possible

  packages:
    gcc:
      externals:
      - spec: gcc@8.5.0 languages:='c,c++,fortran'
        prefix: /usr
        extra_attributes:
          compilers:
            c: /usr/bin/gcc
            cxx: /usr/bin/g++
            fortran: /usr/bin/gfortran
